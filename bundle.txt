package app.osmautosquad

import java.text.Normalizer

/** Oyuncuyu yalnızca adla değil; ad + yaş + pozisyon + reyting ile eşleştirir. */
object Matcher {
    private val MARKS = Regex("\\p{M}")
    private val NONLET = Regex("[^a-z]")

    fun n(s: String): String = Normalizer.normalize(s.replace('ı', 'i').replace('İ', 'I'), Normalizer.Form.NFD)
        .replace(MARKS, "").lowercase().replace(NONLET, "")

    fun lev(a: String, b: String): Int {
        val d = IntArray(b.length + 1) { it }
        for (i in 1..a.length) {
            var p = d[0]; d[0] = i
            for (j in 1..b.length) {
                val t = d[j]
                d[j] = minOf(d[j] + 1, d[j - 1] + 1, p + if (a[i - 1] == b[j - 1]) 0 else 1)
                p = t
            }
        }
        return d[b.length]
    }

    fun nameOk(a: String, b: String): Boolean {
        val x = n(a); val y = n(b)
        if (x.isEmpty() || y.isEmpty()) return false
        return x == y || (minOf(x.length, y.length) >= 6 && lev(x, y) <= 1)
    }

    private fun col(c: String, r: Row): Int? = when (c) { "h" -> r.h; "d" -> r.d; "o" -> r.o; else -> null }

    /** Kaleci (col "k") reytingi listede görünmediği için reyting karşılaştırması atlanır. */
    private fun canon(p: String) = if (p == "KLC") "KL" else p

    fun match(t: Pl, rows: List<Row>): List<Row> = rows.filter { r ->
        nameOk(t.name, r.name) && (t.age == 0 || r.age == t.age) && canon(r.pos) == canon(t.pos) &&
            (col(t.col, r)?.let { it == t.rating } ?: true)
    }
}

package app.osmautosquad

import org.json.JSONObject

data class Pl(
    val role: String, val slot: String, val order: Int, val name: String, val age: Int,
    val club: String, val pos: String, val rating: Int, val col: String, val priority: Boolean
) {
    val key: String get() = "$order|$name|$age"
}

data class Squad(val formation: String, val target: Int, val players: List<Pl>)

object SquadIO {
    fun parse(json: String): Squad {
        val o = JSONObject(json)
        require(o.optInt("version") == 1) { "Desteklenmeyen JSON sürümü" }
        val arr = o.getJSONArray("players")
        val list = (0 until arr.length()).map { i ->
            val q = arr.getJSONObject(i)
            Pl(q.getString("role"), q.getString("slot"), q.getInt("order"), q.getString("name"),
                q.getInt("age"), q.optString("club", ""), q.getString("pos"), q.getInt("rating"),
                q.getString("col"), q.optBoolean("priority", false))
        }
        val s = Squad(o.getString("formation"), o.getInt("target"), list)
        validate(s)
        return s
    }

    /** Web uygulamasındaki kurallarla aynı; otomasyon başlamadan önce çalışır. */
    fun validate(s: Squad) {
        require(s.players.size == 20) { "Toplam oyuncu 20 olmalı (${s.players.size})" }
        require(s.players.count { it.role == "XI" } == 11) { "İlk 11 sayısı 11 olmalı" }
        require(s.players.count { it.role == "SUB" } == 9) { "Yedek sayısı 9 olmalı" }
        require(s.players.map { it.key }.toSet().size == 20) { "Çift oyuncu var" }
        require(s.target in 1..100) { "Hedef reyting 1-100 olmalı" }
        s.players.filter { it.role == "XI" }.forEach {
            require(it.rating == s.target) { "${it.name}: ilk 11 reytingi hedefle aynı değil" }
        }
        val grp = { pos: String ->
            when (pos) { "KL", "KLC" -> "GK"; "SLB", "STP", "SĞB" -> "DF"; "SF", "SLA", "SĞA" -> "FW"; else -> "MD" }
        }
        val subs = s.players.filter { it.role == "SUB" }
        require(subs.count { grp(it.pos) == "GK" } == 2 && subs.count { grp(it.pos) == "DF" } == 3 &&
            subs.count { grp(it.pos) == "MD" } == 2 && subs.count { grp(it.pos) == "FW" } == 2) {
            "Yedek dağılımı 2 KL, 3 DEF, 2 ORT, 2 FOR olmalı"
        }
        val os = s.players.filter { it.role == "XI" && it.pos == "OS" }
        if (os.size == 2) {
            require(os.count { it.age in 20..29 } == 1 && os.count { it.age >= 31 } == 1) { "OS yaş kuralı sağlanmıyor" }
        }
        require(s.players.none { it.role == "SUB" && it.pos == "SĞA" && it.rating == s.target }) {
            "Hedef reytingli SĞA yedekte olamaz"
        }
    }
}

package app.osmautosquad

import android.graphics.Bitmap
import android.graphics.Rect
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlin.coroutines.resume

data class Ln(val text: String, val box: Rect)
data class TRow(val text: String, val y: Int)

object Ocr {
    private val rec by lazy { TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS) }
    private val CNT = Regex("""(\d{1,2})\s*/\s*20""")

    suspend fun lines(b: Bitmap): List<Ln>? = suspendCancellableCoroutine<List<Ln>?> { c ->
        rec.process(InputImage.fromBitmap(b, 0))
            .addOnSuccessListener { r ->
                c.resume(r.textBlocks.flatMap { it.lines }.mapNotNull { l -> l.boundingBox?.let { Ln(l.text, it) } })
            }
            .addOnFailureListener { c.resume(null) }
    }

    /** Aynı yatay hizadaki hücreleri tek satırda birleştirir. */
    fun rows(ls: List<Ln>): List<TRow> {
        if (ls.isEmpty()) return emptyList()
        val sorted = ls.sortedBy { it.box.centerY() }
        val tol = (sorted.map { it.box.height() }.sorted()[sorted.size / 2] * 0.6).toInt().coerceAtLeast(6)
        val groups = mutableListOf<MutableList<Ln>>()
        for (l in sorted) {
            val g = groups.lastOrNull()
            if (g != null && Math.abs(l.box.centerY() - g.map { it.box.centerY() }.average()) < tol) g.add(l)
            else groups.add(mutableListOf(l))
        }
        return groups.map { g ->
            TRow(g.sortedBy { it.box.left }.joinToString(" ") { it.text }, g.map { it.box.centerY() }.average().toInt())
        }
    }

    /** Üst çubuktaki "n / 20" sayacı; seçimlerin doğrulanması bununla yapılır. */
    fun counter(ls: List<Ln>, h: Int): Int? {
        val top = ls.filter { it.box.centerY() < h * 0.15 }.sortedBy { it.box.left }.joinToString(" ") { it.text }
        return CNT.find(top)?.groupValues?.get(1)?.toInt()
    }

    fun isList(ls: List<Ln>): Boolean {
        val t = ls.joinToString(" ") { it.text }
        return t.contains("Fiyat") && t.contains("Oyuncu")
    }
}

/**
 * AŞAMA 1-3: oyuncu listesini oku, oyuncuyu eşleştir, "+" bas, sayaçla doğrula.
 * Saha yerleşimi (AŞAMA 4) ve yedek akışı için OSM'de oyuncu eklendikten sonraki ekranlar gerekir; henüz yok.
 */
class OsmService : AccessibilityService() {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    private var job: Job? = null
    private var wm: WindowManager? = null
    private var panel: LinearLayout? = null
    private lateinit var info: TextView
    private lateinit var prog: TextView
    private var squad: Squad? = null
    private val done = linkedSetOf<String>()
    private var lastShot = 0L
    private var current = ""

    private val prefs get() = getSharedPreferences("osm", Context.MODE_PRIVATE)
    private val targetPkg get() = prefs.getString("pkg", "com.gamebasics.osm") ?: "com.gamebasics.osm"
    private val waitMs get() = prefs.getLong("waitMs", 900L)
    private val retries get() = prefs.getInt("retries", 2)
    private val plusX get() = prefs.getFloat("plusX", 0.924f)   // "+" düğmesinin ekran genişliğine oranı (kalibrasyon yedeği)

    private var evPkg: String? = null
    override fun onAccessibilityEvent(e: AccessibilityEvent?) {
        val p = e?.packageName?.toString() ?: return
        if (p != packageName && !p.startsWith("com.android.systemui") && !p.contains("inputmethod") && !p.contains("keyboard")) evPkg = p
    }
    /** Ön plandaki uygulama: önce aktif pencere, olmazsa son pencere olayı. */
    private fun fgPkg(): String? {
        val r = rootInActiveWindow?.packageName?.toString()
        return if (r != null && r != packageName) r else evPkg
    }
    override fun onInterrupt() { stop() }
    override fun onDestroy() { scope.cancel(); panel?.let { wm?.removeView(it) }; super.onDestroy() }

    override fun onServiceConnected() {
        super.onServiceConnected()
        wm = getSystemService(Context.WINDOW_SERVICE) as WindowManager
        val p = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL; setBackgroundColor(0xE6102040.toInt()); setPadding(16, 12, 16, 12) }
        info = TextView(this).apply { setTextColor(-1); textSize = 12f }
        prog = TextView(this).apply { setTextColor(-1); textSize = 14f }
        val row = LinearLayout(this).apply { orientation = LinearLayout.HORIZONTAL }
        fun b(t: String, f: () -> Unit) { row.addView(Button(this).apply { text = t; textSize = 10f; setOnClickListener { f() } }) }
        b("BAŞLAT") { start() }
        b("DURAKLAT") { pause("Kullanıcı duraklattı") }
        b("DEVAM") { start() }
        b("DURDUR") { stop() }
        val row2 = LinearLayout(this).apply { orientation = LinearLayout.HORIZONTAL }
        row2.addView(Button(this).apply { text = "TANI"; textSize = 10f; setOnClickListener { diagnose() } })
        p.addView(info); p.addView(prog); p.addView(row); p.addView(row2)
        val lp = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT, WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY, WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE,
            PixelFormat.TRANSLUCENT).apply { gravity = Gravity.TOP or Gravity.START; y = 80 }
        wm!!.addView(p, lp); panel = p
        load(); render("Hazır. OSM'de oyuncu listesini aç, sonra BAŞLAT.")
    }

    private fun load() {
        squad = prefs.getString("squad", null)?.let { runCatching { SquadIO.parse(it) }.getOrNull() }
        done.clear()
        val a = JSONArray(prefs.getString("done", "[]"))
        for (i in 0 until a.length()) done.add(a.getString(i))
    }
    private fun save() { prefs.edit().putString("done", JSONArray(done.toList()).toString()).apply() }

    private fun render(msg: String) {
        val s = squad
        info.text = (if (s != null) "${s.formation} · hedef ${s.target}" else "Kadro yok") +
            (if (current.isNotEmpty()) "\nŞu an: $current" else "") + "\n$msg"
        prog.text = "${done.size}/20"
    }

    private fun start() {
        load()
        if (squad == null) { render("Önce uygulamada kadro JSON'unu kaydet."); return }
        if (job?.isActive == true) return
        job = scope.launch { runAll() }
    }
    private fun pause(why: String) { job?.cancel(); render("DURAKLADI: $why") }
    private fun stop() { job?.cancel(); current = ""; render("Durduruldu. İlerleme saklandı.") }
    private fun fail(msg: String): Boolean { pause(msg); return false }

    private suspend fun runAll() {
        val s = squad ?: return
        for (p in s.players) {
            if (p.key in done) continue
            current = "${p.name} (${p.slot} · ${p.pos})"
            render("Aranıyor…")
            if (!step(p)) return
        }
        current = ""
        val c = capture()
        val n = c?.let { Ocr.counter(it.lines, it.h) }
        render(if (n == 20) "Sayaç 20/20 doğrulandı. (Saha yerleşimi bu sürümde otomatik DEĞİL.)"
               else "Tüm oyuncular işlendi ama sayaç 20 okunamadı/doğrulanamadı ($n).")
    }

    private suspend fun step(p: Pl): Boolean {
        var scrolls = 0; var lastSig = ""; var searchTries = 0; var typed = false
        if (findEditable() != null) {
            render("Yazılıyor: ${p.name}")
            if (!typeName(p.name)) return fail("Arama kutusuna yazılamadı/doğrulanamadı: ${p.name}")
            typed = true
            delay(waitMs)
        }
        while (true) {
            currentCoroutineContext().ensureActive()
            val cap = capture() ?: return fail("Ekran görüntüsü/OCR alınamadı")
            if (fgPkg() != targetPkg) return fail("OSM ön planda değil (görülen: ${fgPkg()}, beklenen: $targetPkg)")
            if (!Ocr.isList(cap.lines)) return fail("Oyuncu listesi ekranı tanınmadı")
            val before = Ocr.counter(cap.lines, cap.h) ?: return fail("n/20 sayacı okunamadı")
            if (before >= 20) return fail("Sayaç zaten 20/20")
            val rows = Ocr.rows(cap.lines).mapNotNull { r -> Parser.row(r.text, r.y) }
            val m = Matcher.match(p, rows)
            if (m.size > 1) return fail("Belirsiz eşleşme: ${p.name}")
            if (m.size == 1) {
                if (!gesture(Path().apply { moveTo(cap.w * plusX, m[0].y.toFloat()) }, 60)) return fail("Dokunuş gönderilemedi")
                delay(900)
                val c2 = capture() ?: return fail("Doğrulama görüntüsü alınamadı")
                val after = Ocr.counter(c2.lines, c2.h)
                if (after == before + 1) { done.add(p.key); save(); render("✔ eklendi"); if (typed) clearSearch(); return true }
                return fail("Seçim doğrulanamadı (sayaç $before→$after): ${p.name}")
            }
            if (typed && searchTries < retries) { searchTries++; delay(waitMs); continue }
            val sig = rows.joinToString { it.name }
            if (sig == lastSig || ++scrolls > 40) return fail("Listede bulunamadı: ${p.name}")
            lastSig = sig
            gesture(Path().apply { moveTo(cap.w * 0.5f, cap.h * 0.8f); lineTo(cap.w * 0.5f, cap.h * 0.4f) }, 450)
            delay(700)
        }
    }

    private fun findEditable(): AccessibilityNodeInfo? {
        val root = rootInActiveWindow ?: return null
        var found: AccessibilityNodeInfo? = null
        fun walk(n: AccessibilityNodeInfo?) {
            if (n == null || found != null) return
            if (n.isVisibleToUser && (n.isEditable || n.className?.toString()?.contains("EditText") == true)) { found = n; return }
            for (i in 0 until n.childCount) walk(n.getChild(i))
        }
        walk(root)
        return found
    }

    /** Arama kutusuna yazar ve metnin gerçekten yazıldığını okuyarak doğrular. */
    private suspend fun typeName(name: String): Boolean {
        for (attempt in 0..retries) {
            val n = findEditable()
            if (n != null) {
                n.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
                n.performAction(AccessibilityNodeInfo.ACTION_CLICK)
                delay(300)
                val args = Bundle()
                args.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, name)
                val ok = n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
                delay(300)
                n.refresh()
                if (ok && n.text?.toString()?.trim().equals(name, ignoreCase = true)) {
                    n.performAction(AccessibilityNodeInfo.AccessibilityAction.ACTION_IME_ENTER.id)
                    return true
                }
            }
            delay(waitMs)
        }
        return false
    }

    private fun clearSearch() {
        val n = findEditable() ?: return
        val a = Bundle()
        a.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, "")
        n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, a)
    }

    private fun kbVisible(): Boolean = windows.any { it.type == AccessibilityWindowInfo.TYPE_INPUT_METHOD }

    /** OSM ekranının erişilebilirlik ağacını inceler; arama kutusu görünüyor mu gösterir. */
    private fun diagnose() {
        val root = rootInActiveWindow
        if (root == null) { render("TANI: aktif pencere okunamadı (kök yok)"); return }
        var nodes = 0; var edits = 0; var clicks = 0
        val sb = StringBuilder()
        fun walk(n: AccessibilityNodeInfo?, d: Int) {
            if (n == null || nodes > 3000) return
            nodes++
            val ed = n.isEditable || n.className?.toString()?.contains("EditText") == true
            if (ed) edits++
            if (n.isClickable) clicks++
            val t = n.text?.toString() ?: n.contentDescription?.toString()
            if ((ed || t != null) && sb.length < 4000) {
                sb.append("  ".repeat(d.coerceAtMost(10))).append(n.className?.toString()?.substringAfterLast('.'))
                    .append(if (ed) " [DUZENLENEBILIR]" else "").append(": ").append(t ?: "").append("\n")
            }
            for (i in 0 until n.childCount) walk(n.getChild(i), d + 1)
        }
        walk(root, 0)
        val head = "pkg=${root.packageName} dugum=$nodes duzenlenebilir=$edits tiklanabilir=$clicks klavye=${kbVisible()}"
        val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
        cm.setPrimaryClip(ClipData.newPlainText("tani", head + "\n" + sb))
        render("TANI: $head\n(ayrıntı panoya kopyalandı)")
    }

    private suspend fun gesture(path: Path, dur: Long): Boolean = suspendCancellableCoroutine<Boolean> { c ->
        val g = GestureDescription.Builder().addStroke(GestureDescription.StrokeDescription(path, 0, dur)).build()
        val ok = dispatchGesture(g, object : AccessibilityService.GestureResultCallback() {
            override fun onCompleted(d: GestureDescription?) { c.resume(true) }
            override fun onCancelled(d: GestureDescription?) { c.resume(false) }
        }, null)
        if (!ok) c.resume(false)
    }

    private suspend fun capture(): Cap? {
        val wait = 1200 - (System.currentTimeMillis() - lastShot)
        if (wait > 0) delay(wait)
        panel?.visibility = View.INVISIBLE        // panel metni OCR'ı bozmasın
        delay(200)
        val bmp: Bitmap? = try {
            suspendCancellableCoroutine<Bitmap?> { c ->
                takeScreenshot(Display.DEFAULT_DISPLAY, mainExecutor, object : AccessibilityService.TakeScreenshotCallback {
                    override fun onSuccess(r: AccessibilityService.ScreenshotResult) {
                        val hb = r.hardwareBuffer
                        val b = Bitmap.wrapHardwareBuffer(hb, r.colorSpace)?.copy(Bitmap.Config.ARGB_8888, false)
                        hb.close(); c.resume(b)
                    }
                    override fun onFailure(code: Int) { c.resume(null) }
                })
            }
        } finally { panel?.visibility = View.VISIBLE; lastShot = System.currentTimeMillis() }
        if (bmp == null) return null
        val ls = Ocr.lines(bmp) ?: return null
        return Cap(ls, bmp.width, bmp.height)
    }

package app.osmautosquad

import java.util.Locale

data class Row(val name: String, val age: Int, val club: String, val pos: String,
               val h: Int, val d: Int, val o: Int, val y: Int = 0)

/** Web uygulamasındaki OCR ayrıştırıcının (parseOCR) Kotlin karşılığı; tek bir satır metnini çözer. */
object Parser {
    private val RX = Regex("""([^\d()]+?)\s*\((\d{1,2})\)\s*([^()]+?)\s+([A-ZĞÖÜİŞÇ0]{1,3})\s+(\d{1,3})\s+(\d{1,3})\s+(\d{1,3})""")
    private val LEAD = Regex("^[^A-Za-zÀ-ÿĞğİıŞşÇçÖöÜü]+")
    private val WS = Regex("\\s+")

    fun pos(t: String): String = t.uppercase(Locale("tr")).replace('0', 'O')
        .replace("SGA", "SĞA").replace("SGK", "SĞK").replace("SGB", "SĞB")

    fun row(text: String, y: Int): Row? {
        val m = RX.find(text) ?: return null
        val g = m.groupValues
        val name = g[1].replace(LEAD, "").replace(WS, " ").trim()
        return Row(name, g[2].toInt(), g[3].replace(WS, " ").trim(), pos(g[4]),
            g[5].toInt(), g[6].toInt(), g[7].toInt(), y)
    }
}
