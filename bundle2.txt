fun copyText(ctx: Context, t: String) {
    val cm = ctx.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
    cm.setPrimaryClip(ClipData.newPlainText("oyuncu", t))
    Toast.makeText(ctx, "Kopyalandı: $t", Toast.LENGTH_SHORT).show()
}

fun openOsm(ctx: Context) {
    for (pkg in listOf("com.gamebasics.osm", "com.pengyou.cloneapp")) {
        val i = ctx.packageManager.getLaunchIntentForPackage(pkg)
        if (i != null) {
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            ctx.startActivity(i)
            return
        }
    }
    try {
        val i = Intent(Intent.ACTION_VIEW, Uri.parse("https://play.google.com/store/apps/details?id=com.gamebasics.osm"))
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        ctx.startActivity(i)
    } catch (e: Exception) {
        Toast.makeText(ctx, "OSM açılamadı", Toast.LENGTH_SHORT).show()
    }
}

@Composable
fun App(st: SquadState) {
    var tab by remember { mutableStateOf(0) }
    var editSlot by remember { mutableStateOf<Slot?>(null) }
    var pick by remember { mutableStateOf(st.formationId == null) }

    if (pick || st.formation == null) {
        FormationScreen(st) { pick = false }
    } else {
        Scaffold(
            topBar = { Header(st) },
            bottomBar = {
                NavigationBar {
                    val items = listOf("Saha" to "⚽", "Yedekler" to "🪑", "Liste" to "✅", "JSON" to "💾")
                    items.forEachIndexed { i, it ->
                        NavigationBarItem(
                            selected = tab == i, onClick = { tab = i },
                            icon = { Text(it.second) }, label = { Text(it.first) }
                        )
                    }
                }
            }
        ) { pad ->
            Box(Modifier.padding(pad)) {
                when (tab) {
                    0 -> PitchScreen(st, onPick = { pick = true }, onTap = { editSlot = it })
                    1 -> SubsScreen(st, onTap = { editSlot = it })
                    2 -> ChecklistScreen(st, onEdit = { editSlot = it })
                    else -> JsonScreen(st)
                }
            }
        }
    }
    editSlot?.let { s -> PlayerDialog(st, s) { editSlot = null } }
}

@Composable
fun FormationScreen(st: SquadState, onDone: () -> Unit) {
    var confirm by remember { mutableStateOf<Formation?>(null) }
    var targetText by remember { mutableStateOf(st.target.toString()) }
    Column(Modifier.fillMaxSize().verticalScroll(rememberScrollState()).padding(16.dp)) {
        Text("Diziliş seç", fontSize = 24.sp, fontWeight = FontWeight.Bold)
        Spacer(Modifier.height(8.dp))
        OutlinedTextField(
            value = targetText,
            onValueChange = { t ->
                targetText = t
                val v = t.toIntOrNull()
                if (v != null && v in 1..100) st.setTarget(v)
            },
            label = { Text("Hedef reyting (1-100)") }, singleLine = true,
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
            modifier = Modifier.fillMaxWidth()
        )
        Spacer(Modifier.height(12.dp))
        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
            for (f in Formations.all) {
                Card(
                    Modifier.weight(1f).clickable {
                        val hasXi = st.players.keys.any { it.startsWith("xi") }
                        if (st.formationId != null && st.formationId != f.id && hasXi) confirm = f
                        else { st.setFormation(f.id); onDone() }
                    },
                    border = if (st.formationId == f.id) androidx.compose.foundation.BorderStroke(3.dp, Color(0xFF0D47A1)) else null
                ) {
                    Column(Modifier.padding(10.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                        MiniPitch(f)
                        Spacer(Modifier.height(6.dp))
                        Text(f.id, fontSize = 20.sp, fontWeight = FontWeight.Bold)
                    }
                }
            }
        }
        if (st.formation != null) {
            Spacer(Modifier.height(16.dp))
            Button(onClick = onDone, modifier = Modifier.fillMaxWidth()) { Text("Mevcut dizilişle devam: ${st.formationId}") }
        }
    }
    confirm?.let { f ->
        AlertDialog(
            onDismissRequest = { confirm = null },
            title = { Text("Diziliş değişsin mi?") },
            text = { Text("İlk 11'deki oyuncular silinir (yedekler kalır).") },
            confirmButton = { TextButton(onClick = { st.setFormation(f.id); confirm = null; onDone() }) { Text("Değiştir") } },
            dismissButton = { TextButton(onClick = { confirm = null }) { Text("İptal") } }
        )
    }
}

@Composable
fun ChecklistScreen(st: SquadState, onEdit: (Slot) -> Unit) {
    val ctx = LocalContext.current
    val order = st.orderedSlots()
    val missing = st.missingLabels()
    Column(Modifier.fillMaxSize()) {
        if (missing.isNotEmpty()) {
            Column(Modifier.padding(16.dp)) {
                Text("Kontrol listesi için 20/20 gerekli.", fontWeight = FontWeight.Bold)
                Text("Eksik: ${missing.joinToString()}", color = Color(0xFFD32F2F))
            }
        } else {
            val added = order.count { st.done[it.id] == "added" }
            val skipped = order.count { st.done[it.id] == "skipped" }
            Column(Modifier.padding(10.dp)) {
                Text("OSM'ye eklenen: $added/20 · atlanan: $skipped", fontWeight = FontWeight.Bold)
                if (added == 20) Text("KADRO HAZIRLANDI ✔", color = Color(0xFF2E7D32), fontSize = 20.sp, fontWeight = FontWeight.Bold)
                Text("Adı kopyala → OSM'de 'Oyuncu ara'ya yapıştır → ekle → buradan işaretle. Hiçbir şey otomatik yapılmaz.", fontSize = 12.sp, color = Color.Gray)
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    Button(onClick = { openOsm(ctx) }) { Text("OSM'yi aç") }
                    OutlinedButton(onClick = { st.resetProgress() }) { Text("İlerlemeyi sıfırla") }
                }
            }
            Column(Modifier.weight(1f).verticalScroll(rememberScrollState()).padding(horizontal = 10.dp)) {
                order.forEachIndexed { i, s ->
                    val p = st.players[s.id] ?: return@forEachIndexed
                    val status = st.done[s.id]
                    Card(Modifier.fillMaxWidth().padding(vertical = 3.dp)) {
                        Column(Modifier.padding(10.dp)) {
                            Text("${i + 1}. ${p.name}", fontSize = 18.sp, fontWeight = FontWeight.Bold)
                            Text("${s.label} · ${p.age} yaş · reyting ${p.rating}" +
                                (if (status == "added") " · ✔ eklendi" else if (status == "skipped") " · atlandı" else ""),
                                fontSize = 13.sp)
                            Row(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                                OutlinedButton(onClick = { copyText(ctx, p.name) }) { Text("Kopyala") }
                                Button(onClick = { st.toggle(s.id, "added") }) { Text(if (status == "added") "✔ Eklendi" else "Eklendi") }
                            }
                            Row {
                                TextButton(onClick = { st.toggle(s.id, "skipped") }) { Text(if (status == "skipped") "Atlamayı kaldır" else "Atla") }
                                TextButton(onClick = { onEdit(s) }) { Text("Düzenle") }
                            }
                        }
                    }
                }
                Spacer(Modifier.height(12.dp))
            }
        }
    }
}

                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {

                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number), modifier = Modifier.weight(1f))
                    OutlinedTextField(value = rating, onValueChange = { rating = it }, label = { Text("Reyting") }, singleLine = true,

                }
                err?.let { Text(it, color = Color(0xFFD32F2F), fontSize = 13.sp) }
                if (cur != null) TextButton(onClick = { st.remove(slot.id); onClose() }) { Text("Oyuncuyu sil") }
            }
        },
        confirmButton = {
            Button(onClick = {
                val a = age.toIntOrNull()
                val r = rating.toIntOrNull()
                when {
                    name.isBlank() -> err = "İsim gerekli"
                    a == null || a !in 1..60 -> err = "Yaş (1-60) gir"
                    r == null || r !in 1..100 -> err = "Reyting (1-100) gir"
                    else -> {
                        val e = st.put(slot.id, Pl(name.trim(), a, r))
                        if (e == null) onClose() else err = e
                    }
                }
            }) { Text("Kaydet") }
        },
        dismissButton = { TextButton(onClick = onClose) { Text("İptal") } }
    )
}

fun fold(s: String): String =
    Normalizer.normalize(s.replace('ı', 'i').replace('İ', 'I'), Normalizer.Form.NFD)
        .replace(Regex("\\p{M}"), "").lowercase().replace(Regex("[^a-z0-9]"), "")

/** Tüm kadro durumu + yerel kayıt (SharedPreferences, JSON). */
class SquadState(ctx: Context) {
    private val prefs = ctx.getSharedPreferences("osm_squad", Context.MODE_PRIVATE)

    var formationId by mutableStateOf<String?>(null)
    var target by mutableStateOf(80)
    val players = mutableStateMapOf<String, Pl>()
    val done = mutableStateMapOf<String, String>()   // yuva -> "added" | "skipped"

    init {
        prefs.getString("json", null)?.let { fromJson(it) }
    }

    val formation: Formation? get() = Formations.all.find { it.id == formationId }

    fun slotById(id: String): Slot? = formation?.slots?.find { it.id == id } ?: Formations.SUBS.find { it.id == id }

    fun setFormation(id: String) {
        if (formationId != id) {
            val xiKeys = players.keys.filter { it.startsWith("xi") }
            xiKeys.forEach { players.remove(it); done.remove(it) }
        }
        formationId = id
        persist()
    }

    fun setTarget(v: Int) { target = v; persist() }

    /** Hata mesajı döner; başarılıysa null. Veri hiçbir zaman otomatik üretilmez. */
    fun put(id: String, p: Pl): String? {
        val other = players.entries.firstOrNull { it.key != id && fold(it.value.name) == fold(p.name) }
        if (other != null) return "Bu oyuncu zaten kadroda (${slotById(other.key)?.label ?: other.key})"
        if (!players.containsKey(id) && players.size >= 20) return "Kadro 20/20: önce birini sil"
        players[id] = p
        persist()
        return null
    }

    fun remove(id: String) { players.remove(id); done.remove(id); persist() }

    fun toggle(id: String, status: String) {
        if (done[id] == status) done.remove(id) else done[id] = status
        persist()
    }

    fun resetProgress() { done.clear(); persist() }

    /** İlk 11 (kaleci, defans, orta saha, forvet sırasıyla) + 9 yedek. */
    fun orderedSlots(): List<Slot> {
        val xi = formation?.slots ?: emptyList()
        val sorted = xi.sortedBy { Formations.GROUP_ORDER.indexOf(it.group) }
        return sorted + Formations.SUBS
    }

    fun missingLabels(): List<String> =
        orderedSlots().filter { !players.containsKey(it.id) }.map { it.label }

    /** Uyarılar bilgilendiricidir, engellemez. İlk OS yuvası 31+ yaş kuralına tabidir. */
    fun warnings(): Map<String, String> {
        val f = formation ?: return emptyMap()
        val w = HashMap<String, String>()
        val oldId = f.slots.firstOrNull { it.label == "OS" }?.id
        for (s in f.slots) {
            val p = players[s.id] ?: continue
            val m = mutableListOf<String>()
            if (p.rating != target) m.add("reyting ≠ $target")
            if (s.id == oldId) { if (p.age < 31) m.add("bu OS 31+ olmalı") }
            else if (p.age !in 20..29) m.add("yaş 20-29 değil")
            if (m.isNotEmpty()) w[s.id] = m.joinToString(", ")
        }
        return w
    }

    fun toJson(): String {
        val o = JSONObject()
        o.put("version", 2)
        o.put("formation", formationId ?: JSONObject.NULL)
        o.put("target", target)
        val po = JSONObject()
        for ((k, v) in players) po.put(k, JSONObject().put("n", v.name).put("a", v.age).put("r", v.rating))
        o.put("players", po)
        val d = JSONObject()
        for ((k, v) in done) d.put(k, v)
        o.put("done", d)
        return o.toString()
    }

    /** Hata mesajı döner; başarılıysa null ve durumu değiştirir. */
    fun fromJson(s: String): String? {
        try {
            val o = JSONObject(s)
            if (o.optInt("version") != 2) return "Desteklenmeyen sürüm (version 2 bekleniyor)"
            val fid: String? = if (o.isNull("formation")) null else o.getString("formation")
            var f: Formation? = null
            if (fid != null) {
                f = Formations.all.find { it.id == fid }
                if (f == null) return "Bilinmeyen diziliş: $fid"
            }
            val valid = (f?.slots?.map { it.id } ?: emptyList<String>()) + Formations.SUBS.map { it.id }
            val po = o.getJSONObject("players")
            val tmp = LinkedHashMap<String, Pl>()
            val names = HashSet<String>()
            for (k in po.keys()) {
                if (k !in valid) return "Geçersiz yuva: $k"
                val q = po.getJSONObject(k)
                val pl = Pl(q.getString("n").trim(), q.getInt("a"), q.getInt("r"))
                if (pl.name.isEmpty()) return "Boş isim: $k"
                if (pl.rating !in 1..100) return "Reyting 1-100 olmalı: ${pl.name}"
                if (pl.age !in 1..60) return "Yaş 1-60 olmalı: ${pl.name}"
                if (!names.add(fold(pl.name))) return "Tekrarlanan oyuncu: ${pl.name}"
                tmp[k] = pl
            }
            if (tmp.size > 20) return "20'den fazla oyuncu"
            val tmpDone = LinkedHashMap<String, String>()
            val d = o.optJSONObject("done")
            if (d != null) {
                for (k in d.keys()) {
                    val v = d.getString(k)
                    if (tmp.containsKey(k) && (v == "added" || v == "skipped")) tmpDone[k] = v
                }
            }
            formationId = fid
            target = o.optInt("target", 80).coerceIn(1, 100)
            players.clear(); players.putAll(tmp)
            done.clear(); done.putAll(tmpDone)
            persist()
            return null
        } catch (e: Exception) {
            return "JSON okunamadı: ${e.message}"
        }
    }

    private fun persist() { prefs.edit().putString("json", toJson()).apply() }
}
